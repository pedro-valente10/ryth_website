import os

output_file = 'projeto_consolidado.txt'
ignore_dirs = {'node_modules', '.git', 'dist', 'build', '.vscode'}
ignore_exts = {'.png', '.jpg', '.jpeg', '.svg', '.ico', '.woff', '.woff2', '.ttf', '.lock'}

with open(output_file, 'w', encoding='utf-8') as outfile:
    for root, dirs, files in os.walk('.'):
        dirs[:] = [d for d in dirs if d not in ignore_dirs]
        for file in files:
            if file == output_file or file == 'package-lock.json' or any(file.endswith(ext) for ext in ignore_exts):
                continue
            path = os.path.join(root, file)
            outfile.write(f'========================================\n')
            outfile.write(f'ARQUIVO: {path}\n')
            outfile.write(f'========================================\n\n')
            try:
                with open(path, 'r', encoding='utf-8') as f:
                    outfile.write(f.read())
            except Exception:
                outfile.write('[Arquivo não lido ou binário]\n')
            outfile.write('\n\n')

print('Concluído! Arquivo gerado: projeto_consolidado.txt')

# RYTH Website

Site institucional da RYTH, construído com **Vite + TypeScript**, sem framework de UI (HTML/CSS/TS puro).

## Stack

- [Vite](https://vitejs.dev/) — bundler e dev server
- TypeScript
- HTML + CSS puro (sem framework de UI)

## Pré-requisitos

- Node.js (LTS mais recente recomendado)
- npm

## Estrutura de pastas

import "./styles/reset.css";
import "./styles/global.css";

import "./components/navbar/navbar.css";

import { renderNavbar } from "./components/navbar/navbar";
import { renderFooter } from "./components/footer/footer";

/**
 * Remove o bloqueio de visibilidade assim que os componentes e estilos terminam de carregar.
 */
function revelarPagina(): void {
  document.documentElement.classList.add("ready");
}

/**
 * Normaliza os caminhos de URL para marcar o link ativo da Navbar.
 */
function markActiveLink(): void {
  const normalize = (path: string): string =>
    path.replace(/\/index\.html$/, "/").replace(/\/$/, "") || "/";

  const current = normalize(location.pathname);

  document
    .querySelectorAll<HTMLAnchorElement>(".navbar__link a")
    .forEach((link) => {
      const href = normalize(new URL(link.href, location.origin).pathname);

      if (href === current) {
        link.setAttribute("aria-current", "page");
      } else {
        link.removeAttribute("aria-current");
      }
    });
}

/**
 * Inicializa a Navbar, o Footer e libera a exibição limpa da página.
 */
async function init(): Promise<void> {
  try {
    renderFooter();
    await renderNavbar(); // espera a sessão do usuário e monta a navbar
    markActiveLink();     // marca o link ativo com a navbar pronta
  } catch (error) {
    console.error("Erro ao carregar os componentes globais:", error);
  } finally {
    revelarPagina();      // revela a página perfeitamente montada sem vazamentos de HTML
  }
}

if (document.readyState === "loading") {
  document.addEventListener("DOMContentLoaded", init);
} else {
  init();
}

let modalInjetado = false;

/**
 * Injeta a estrutura HTML do modal no DOM se ainda não existir.

function garantirModalNoDom(): void {
  if (modalInjetado || document.getElementById("modal-container")) return;

/**
 * Exibe o popup com mensagem e configurações customizadas.
 */
export function abrirModal(opcoes: ModalOpcoes): void {
  garantirModalNoDom();

  const modal = document.getElementById("modal-container");
  const overlay = document.getElementById("modal-overlay");
  const closeBtn = document.getElementById("modal-close");
  const confirmBtn = document.getElementById("modal-confirm-btn") as HTMLButtonElement | null;
  const titleEl = document.getElementById("modal-title");
  const messageEl = document.getElementById("modal-message");

  if (!modal || !messageEl) return;

  // Atualiza textos
  if (titleEl) titleEl.textContent = opcoes.titulo || "Aviso";
  messageEl.textContent = opcoes.mensagem;
  if (confirmBtn) confirmBtn.textContent = opcoes.botaoTexto || "Entendi";

  const fechar = () => {
    modal.classList.remove("modal--open");
    modal.setAttribute("aria-hidden", "true");
    setTimeout(() => {
      modal.hidden = true;
    }, 250);
  };

  // Eventos de fechar
  overlay?.addEventListener("click", fechar, { once: true });
  closeBtn?.addEventListener("click", fechar, { once: true });

  if (confirmBtn) {
    confirmBtn.onclick = () => {
      fechar();
      if (opcoes.aoConfirmar) opcoes.aoConfirmar();
    };
  }

  // Exibe com animação
  modal.hidden = false;
  requestAnimationFrame(() => {
    modal.classList.add("modal--open");
    modal.setAttribute("aria-hidden", "false");
    confirmBtn?.focus();
  });
}

// src/services/cms.ts
import { sanityClient, urlFor } from './sanity'

// Retorna null em qualquer falha ou resultado vazio → o HTML padrão permanece
export async function cmsFetch<T = Record<string, any>>(
  query: string,
  params: Record<string, unknown> = {}
): Promise<T | null> {
  if (!sanityClient) return null
  try {
    const data = await sanityClient.fetch<T>(query, params)
    if (data == null || (Array.isArray(data) && data.length === 0)) return null
    return data
  } catch (err) {
    console.error('[CMS]', err)
    return null
  }
}

// Preenche o HTML com os dados:
//   data-cms="campo"                → textContent
//   data-cms="campo" data-cms-attr="href" → atributo
//   data-cms-img="campo"            → <img src> a partir de uma imagem do Sanity
//   data-cms-list="campo" + <template> → lista repetida
export function applyCms(root: ParentNode, data: Record<string, any>): void {
  root.querySelectorAll<HTMLElement>('[data-cms]').forEach((el) => {
    const value = data[el.dataset.cms!]
    if (typeof value !== 'string') return
    const attr = el.dataset.cmsAttr
    if (attr) el.setAttribute(attr, value)
    else el.textContent = value // textContent evita injeção de HTML
  })

  root.querySelectorAll<HTMLImageElement>('[data-cms-img]').forEach((img) => {
    const src = urlFor(data[img.dataset.cmsImg!])
    if (src) img.src = src
  })

  root.querySelectorAll<HTMLElement>('[data-cms-list]').forEach((container) => {
    const items = data[container.dataset.cmsList!]
    const tpl = container.querySelector('template')
    if (!Array.isArray(items) || items.length === 0 || !tpl) return

    Array.from(container.children).forEach((c) => c !== tpl && c.remove())
    items.forEach((item) => {
      const node = tpl.content.cloneNode(true) as DocumentFragment
      applyCms(node, item)
      container.appendChild(node)
    })
  })
}

export const SENHA_MIN = 8;

// Requisitos de senha: a lista alimenta a validação E o texto de ajuda.
// Para exigir mais (letra, número...), é só adicionar um item aqui.
export const REQUISITOS_SENHA = [
  { texto: `ter pelo menos ${SENHA_MIN} caracteres`, ok: (s: string) => s.length >= SENHA_MIN },
  // { texto: "ter pelo menos um número", ok: (s: string) => /\d/.test(s) },
  // { texto: "ter pelo menos uma letra", ok: (s: string) => /[A-Za-z]/.test(s) },
];

export function textoRequisitosSenha(): string {
  return `A senha precisa ${REQUISITOS_SENHA.map((r) => r.texto).join(", ")}.`;
}

export function validarEmail(email: string): string | null {
  if (!email) return "Informe seu e-mail.";
  if (!/^[^\s@]+@[^\s@]+\.[^\s@]{2,}$/.test(email)) {
    return "Esse e-mail não parece válido. Use o formato nome@exemplo.com.";
  }
  return null;
}

export function validarSenhaCadastro(senha: string): string | null {
  if (!senha) return "Crie uma senha.";
  const faltando = REQUISITOS_SENHA.filter((r) => !r.ok(senha)).map((r) => r.texto);
  return faltando.length ? `A senha precisa ${faltando.join(" e ")}.` : null;
}

const MENSAGENS_POR_CODIGO: Record<string, string> = {
  invalid_credentials: "E-mail ou senha incorretos. Confira os dados e tente de novo.",
  email_not_confirmed: "Você ainda não confirmou seu e-mail. Procure o código na sua caixa de entrada (e no spam).",
  user_already_exists: "Este e-mail já está cadastrado. Tente fazer login.",
  email_exists: "Este e-mail já está cadastrado. Tente fazer login.",
  weak_password: textoRequisitosSenha(),
  email_address_invalid: "Esse e-mail não é aceito. Verifique se digitou corretamente.",
  signup_disabled: "Os cadastros estão temporariamente desativados.",
  over_request_rate_limit: "Muitas tentativas seguidas. Aguarde alguns minutos e tente de novo.",
  over_email_send_rate_limit: "Muitos e-mails enviados. Aguarde alguns minutos antes de pedir outro código.",
  otp_expired: "Código inválido ou expirado. Peça um novo código.",
  user_not_found: "Não encontramos uma conta com esse e-mail.",
  same_password: "A nova senha precisa ser diferente da atual.",
};

export function traduzirErroAuth(err: unknown): string {
  const e = err as { code?: string; message?: string; status?: number } | null;

  if (e?.code && MENSAGENS_POR_CODIGO[e.code]) return MENSAGENS_POR_CODIGO[e.code];

  // Fallback por texto (versões antigas do supabase-js não enviam "code")
  const msg = (e?.message ?? "").toLowerCase();
  if (msg.includes("invalid login credentials")) return MENSAGENS_POR_CODIGO.invalid_credentials;
  if (msg.includes("email not confirmed")) return MENSAGENS_POR_CODIGO.email_not_confirmed;
  if (msg.includes("already registered")) return MENSAGENS_POR_CODIGO.user_already_exists;
  if (msg.includes("password should be")) return textoRequisitosSenha();
  if (msg.includes("rate limit") || e?.status === 429) return MENSAGENS_POR_CODIGO.over_request_rate_limit;
  if (msg.includes("token has expired") || msg.includes("invalid token")) return MENSAGENS_POR_CODIGO.otp_expired;
  if (msg.includes("failed to fetch") || msg.includes("network")) {
    return "Não foi possível conectar. Verifique sua internet e tente novamente.";
  }

  return "Algo deu errado. Tente novamente em instantes.";
}

import { supabase } from "./supabase";
import { traduzirErroAuth } from "./erros";
import { ROTAS } from "../config/routes";

export interface Usuario {
  id?: string;
  email: string;
}

/**
 * Retorna o usuário logado (via Supabase ou localStorage de teste)
 */
export async function usuarioAtual(): Promise<Usuario | null> {
  // 1. Sessão ativa no Supabase
  if (supabase) {
    const { data: { session } } = await supabase.auth.getSession();
    if (session?.user?.email) {
      return { id: session.user.id, email: session.user.email };
    }
    return null;
  }

  // 2. Fallback para modo de teste local (só quando o Supabase não está configurado)
  const usuarioTeste = localStorage.getItem("usuario-teste");
  return usuarioTeste ? JSON.parse(usuarioTeste) : null;
}

/**
 * Realiza o login com e-mail e senha
 */
export async function fazerLogin(email: string, senha: string): Promise<Usuario> {
  if (supabase) {
    const { data, error } = await supabase.auth.signInWithPassword({
      email,
      password: senha,
    });

    if (error) {
      throw new Error(traduzirErroAuth(error));
    }

    if (data.user?.email) {
      return { id: data.user.id, email: data.user.email };
    }

    throw new Error("Não foi possível obter os dados do usuário.");
  }

  // Fallback de teste caso o Supabase não esteja conectado
  if (email && senha.length >= 6) {
    const usuarioFake: Usuario = { id: "dev-123", email };
    localStorage.setItem("usuario-teste", JSON.stringify(usuarioFake));
    return usuarioFake;
  }

  throw new Error("E-mail ou senha incorretos.");
}

/**
 * Cria uma conta nova com e-mail e senha
 */
export async function fazerCadastro(
  email: string,
  senha: string
): Promise<{ precisaConfirmar: boolean }> {
  if (!supabase) {
    throw new Error("Cadastro indisponível: o Supabase não está configurado.");
  }

  const { data, error } = await supabase.auth.signUp({
    email,
    password: senha,
    options: { emailRedirectTo: `${window.location.origin}${ROTAS.perfil}` },
  });

  if (error) {
    throw new Error(traduzirErroAuth(error));
  }

  // E-mail já cadastrado: o Supabase não devolve erro, mas retorna identities vazio
  if (data.user && data.user.identities?.length === 0) {
    throw new Error("Este e-mail já está cadastrado. Tente fazer login.");
  }

  // Sem sessão = o Supabase está exigindo confirmação por e-mail
  return { precisaConfirmar: !data.session };
}

/**
 * Confirma o e-mail do novo usuário com o código recebido
 */
export async function confirmarCodigo(email: string, codigo: string): Promise<void> {
  if (!supabase) {
    throw new Error("Supabase não está configurado.");
  }

  const { error } = await supabase.auth.verifyOtp({
    email,
    token: codigo.trim(),
    type: "signup",
  });

  }
}

/**
 * Reenvia o código de confirmação para o e-mail
 */
export async function reenviarCodigo(email: string): Promise<void> {
  if (!supabase) {
    throw new Error("Supabase não está configurado.");
  }

  const { error } = await supabase.auth.resend({ type: "signup", email });

  if (error) {
    throw new Error(traduzirErroAuth(error));
  }
}

/**
 * Encerra a sessão do usuário
 */
export async function fazerLogout(): Promise<void> {
  if (supabase) {
    await supabase.auth.signOut();
  }
  localStorage.removeItem("usuario-teste");
}

import "./perfil.css";
import { usuarioAtual } from "../../services/auth";
import { ROTAS } from "../../config/routes";

export async function carregarPerfil(): Promise<void> {
  // 1. Verifica se o usuário está logado
  const usuario = await usuarioAtual();

  if (!usuario) {
    window.location.href = ROTAS.login;
    return;
  }

  // 2. Atualiza o nome/e-mail no perfil
  const nameEl = document.querySelector<HTMLElement>("#user-name");
  if (nameEl && usuario.email) {
    nameEl.textContent = usuario.email.split("@")[0];
  }

  // 3. Interatividade das abas
  const tabs = document.querySelectorAll<HTMLButtonElement>(".profile__tab");
  tabs.forEach((tab) => {
    tab.addEventListener("click", () => {
      tabs.forEach((t) => t.classList.remove("profile__tab--active"));
      tab.classList.add("profile__tab--active");
    });
  });

  // 4. Ação do botão "editar"
  const btnEditar = document.querySelector<HTMLButtonElement>("#btn-editar");
  if (btnEditar) {
    btnEditar.addEventListener("click", () => {
    window.location.href = ROTAS.edicao;
    });
  }
}

// Executa a função na inicialização do módulo
carregarPerfil();

import "./edicao.css";
import { usuarioAtual } from "../../services/auth";
import { ROTAS } from "../../config/routes";

interface DadosPerfil {
  nome: string;
  bio: string;
  estilos: string[];
  instagram: string;
  tiktok: string;
}

const TAMANHO_MAX_IMAGEM = 5 * 1024 * 1024; // 5 MB

/* --------------------------------------------------------------------------
   Persistência
   Por enquanto salva no localStorage (por usuário). Quando o backend
   estiver pronto, troque APENAS as funções lerPerfil e salvarPerfil.
   -------------------------------------------------------------------------- */
function chaveStorage(email: string): string {
  return `ryth:perfil:${email}`;
}

function lerPerfil(email: string): Partial<DadosPerfil> {
  try {
    const bruto = localStorage.getItem(chaveStorage(email));
    return bruto ? (JSON.parse(bruto) as Partial<DadosPerfil>) : {};
  } catch {
    return {};
  }
}

async function salvarPerfil(email: string, dados: DadosPerfil): Promise<void> {
  localStorage.setItem(chaveStorage(email), JSON.stringify(dados));
}

/* --------------------------------------------------------------------------
   Utilitários
   -------------------------------------------------------------------------- */
function el<T extends HTMLElement>(id: string): T | null {
  return document.getElementById(id) as T | null;
}

function autoAjustarAltura(textarea: HTMLTextAreaElement): void {
  textarea.style.height = "auto";
  textarea.style.height = `${textarea.scrollHeight}px`;
}

function linkValido(valor: string): boolean {
  if (!valor) return true; // campo opcional
  try {
    const url = new URL(valor);
    return url.protocol === "https:" || url.protocol === "http:";
  } catch {
    return false;
  }
}

function mostrarErro(mensagem: string): void {
  const erroEl = el<HTMLParagraphElement>("edit-erro");
  if (!erroEl) return;
  erroEl.textContent = mensagem;
  erroEl.hidden = !mensagem;
}

/* --------------------------------------------------------------------------
   Pré-visualização das fotos
   (apenas visual por enquanto; o upload real depende de um storage)
   -------------------------------------------------------------------------- */
function inicializarPreviewFoto(inputId: string, previewId: string, placeholderId?: string): void {
  const input = el<HTMLInputElement>(inputId);
  const preview = el<HTMLElement>(previewId);
  const placeholder = placeholderId ? el<HTMLElement>(placeholderId) : null;
  if (!input || !preview) return;

  let urlAnterior: string | null = null;

  input.addEventListener("change", () => {
    const arquivo = input.files?.[0];
    if (!arquivo) return;

    if (!arquivo.type.startsWith("image/")) {
      mostrarErro("Selecione um arquivo de imagem.");
      input.value = "";
      return;
    }
    if (arquivo.size > TAMANHO_MAX_IMAGEM) {
      mostrarErro("A imagem deve ter no máximo 5 MB.");
      input.value = "";
      return;
    }

    mostrarErro("");
    if (urlAnterior) URL.revokeObjectURL(urlAnterior);
    urlAnterior = URL.createObjectURL(arquivo);

    preview.style.backgroundImage = `url("${urlAnterior}")`;
    if (placeholder) placeholder.hidden = true;
  });
}

/* --------------------------------------------------------------------------
   Inicialização
   -------------------------------------------------------------------------- */
export async function carregarEdicao(): Promise<void> {
  // 1. Só usuário logado pode editar
  const usuario = await usuarioAtual();
  if (!usuario || !usuario.email) {
    window.location.href = ROTAS.login;
    return;
  }
  const email = usuario.email;

  // 2. Botão voltar
  const btnVoltar = el<HTMLAnchorElement>("btn-voltar");
  if (btnVoltar) btnVoltar.href = ROTAS.perfil;

  // 3. Preenche os campos com os dados salvos (ou valores padrão)
  const salvo = lerPerfil(email);

  const campoNome = el<HTMLInputElement>("campo-nome");
  const campoBio = el<HTMLTextAreaElement>("campo-bio");
  const campoEstilos = el<HTMLTextAreaElement>("campo-estilos");
  const campoInstagram = el<HTMLInputElement>("campo-instagram");
  const campoTiktok = el<HTMLInputElement>("campo-tiktok");

  if (campoNome) campoNome.value = salvo.nome ?? email.split("@")[0];
  if (campoBio) campoBio.value = salvo.bio ?? "";
  if (campoEstilos) campoEstilos.value = (salvo.estilos ?? []).join(", ");
  if (campoInstagram) campoInstagram.value = salvo.instagram ?? "";
  if (campoTiktok) campoTiktok.value = salvo.tiktok ?? "";

  // 4. Textareas crescem conforme o texto
  [campoBio, campoEstilos].forEach((textarea) => {
    if (!textarea) return;
    autoAjustarAltura(textarea);
    textarea.addEventListener("input", () => autoAjustarAltura(textarea));
  });

  // 5. Clicar em qualquer ponto da pílula foca o campo
  document.querySelectorAll<HTMLElement>(".edit__field").forEach((campo) => {
    campo.addEventListener("click", (e) => {
      if (e.target === campo) {
        campo.querySelector<HTMLInputElement | HTMLTextAreaElement>(".edit__input")?.focus();
      }
    });
  });

  // 6. Pré-visualização das fotos
  inicializarPreviewFoto("input-avatar", "avatar-preview", "avatar-placeholder");
  inicializarPreviewFoto("input-cover", "cover-preview");

  // 7. Salvar
  const form = el<HTMLFormElement>("form-edicao");
  const btnSalvar = el<HTMLButtonElement>("btn-salvar");

  form?.addEventListener("submit", async (e) => {
    e.preventDefault();
    mostrarErro("");

    const nome = campoNome?.value.trim() ?? "";
    const bio = campoBio?.value.trim() ?? "";
    const estilos = (campoEstilos?.value ?? "")
      .split(",")
      .map((s) => s.trim())
      .filter(Boolean);
    const instagram = campoInstagram?.value.trim() ?? "";
    const tiktok = campoTiktok?.value.trim() ?? "";

    if (!nome) {
      mostrarErro("Informe um nome de usuário.");
      campoNome?.focus();
      return;
    }
    if (!linkValido(instagram)) {
      mostrarErro("O link do Instagram é inválido. Use o endereço completo, começando com https://");
      campoInstagram?.focus();
      return;
    }
    if (!linkValido(tiktok)) {
      mostrarErro("O link do TikTok é inválido. Use o endereço completo, começando com https://");
      campoTiktok?.focus();
      return;
    }

    if (btnSalvar) btnSalvar.disabled = true;

    try {
      await salvarPerfil(email, { nome, bio, estilos, instagram, tiktok });
      window.location.href = ROTAS.perfil;
    } catch (error) {
      const mensagem =
        error instanceof Error ? error.message : "Não foi possível salvar o perfil. Tente novamente.";
      mostrarErro(mensagem);
      if (btnSalvar) btnSalvar.disabled = false;
    }
  });
}

// Executa a função na inicialização do módulo
carregarEdicao();

import "./competicoes.css";
import { cmsFetch } from "../../services/cms";

type Competicao = {
  titulo: string;
  slug: string;
  ativa?: boolean;
  imagem?: string;
};

const BASE_DETALHE = "/src/pages/competicoes/competicao.html";

function cardHTML(c: Competicao, estado: "active" | "finished") {
  const href = `${BASE_DETALHE}?slug=${encodeURIComponent(c.slug)}`;
  const cta =
    estado === "active" ? `<span class="competicoes__cta">Inscreva-se</span>` : "";
  const img = c.imagem
    ? `<img src="${c.imagem}?w=800&auto=format" alt="${c.titulo}" loading="lazy" />`
    : "";

  return `
    <a class="competicoes__card competicoes__card--${estado}" href="${href}">
      <span class="competicoes__media">${img}${cta}</span>
      <span class="competicoes__name">${c.titulo}</span>
    </a>`;
}

async function renderLista() {
  const gridAtivas = document.querySelector<HTMLElement>("#grid-ativas");
  const gridFinalizadas = document.querySelector<HTMLElement>("#grid-finalizadas");
  if (!gridAtivas || !gridFinalizadas) return;

  const competicoes = await cmsFetch<Competicao[]>(
    `*[_type == "competicao"] | order(ordem asc, _createdAt desc) {
      titulo,
      "slug": slug.current,
      ativa,
      "imagem": imagemCard.asset->url
    }`
  );

  if (!competicoes) {
    console.warn("Competições não carregadas: confira o .env do Sanity.");
    return;
  }

  const ativas = competicoes.filter((c) => c.ativa);
  const finalizadas = competicoes.filter((c) => !c.ativa);

  gridAtivas.innerHTML =
    ativas.map((c) => cardHTML(c, "active")).join("") ||
    "<p>Nenhuma competição ativa no momento.</p>";

  gridFinalizadas.innerHTML =
    finalizadas.map((c) => cardHTML(c, "finished")).join("") ||
    "<p>Nenhuma competição finalizada.</p>";
}

renderLista();

import "./competicao.css";
import { cmsFetch } from "../../services/cms";
import { ROTAS } from "../../config/routes";

type Competicao = {
  titulo: string;
  musicaTitulo?: string;
  musicaDescricao?: string;
  premios?: string;
  prazos?: string;
  estilo?: string;
  banner?: string;
};

function setText(id: string, valor?: string): void {
  const el = document.getElementById(id);
  if (el && valor) el.textContent = valor;
}

function slugAtual(): string | null {
  return new URLSearchParams(window.location.search).get("slug");
}

async function carregarDadosSanity(): Promise<void> {
  const slug = slugAtual();
  if (!slug) return;

  const comp = await cmsFetch<Competicao>(
    `*[_type == "competicao" && slug.current == $slug]{
      titulo,
      musicaTitulo,
      musicaDescricao,
      premios,
      prazos,
      estilo,
      "banner": banner.asset->url
    }`,
    { slug }
  );

  if (!comp) return;

  document.title = `RYTH - ${comp.titulo}`;
  setText("musica-titulo", comp.musicaTitulo);
  setText("musica-descricao", comp.musicaDescricao);
  setText("premios-texto", comp.premios);
  setText("prazos-texto", comp.prazos);
  setText("estilo-texto", comp.estilo);

  const bannerEl = document.querySelector<HTMLImageElement>("#competicao-banner");
  if (bannerEl && comp.banner) {
    bannerEl.src = `${comp.banner}?w=1600&auto=format`;
    bannerEl.alt = comp.titulo;
  }
}

function inicializarBotaoInscricao(): void {
  const btnInscrever = document.querySelector<HTMLButtonElement>("#btn-inscrever");
  if (!btnInscrever) return;

  btnInscrever.addEventListener("click", () => {
    const slug = slugAtual();
    window.location.href = slug ? `${ROTAS.inscricao}?slug=${encodeURIComponent(slug)}` : ROTAS.competicoes;
  });
}

function inicializarInteratividadeVotacao(): void {
  const participantes = document.querySelectorAll<HTMLButtonElement>(".competicao__participante");
  const confirmarNome = document.querySelector<HTMLElement>("#confirmar-nome");
  const btnVotarMain = document.querySelector<HTMLButtonElement>("#btn-votar");
  const btnConfirmarVoto = document.querySelector<HTMLButtonElement>("#btn-confirmar");

  let participanteSelecionado: string | null = null;

  participantes.forEach((p) => {
    p.addEventListener("click", () => {
      participantes.forEach((el) => el.classList.remove("competicao__participante--selecionado"));
      p.classList.add("competicao__participante--selecionado");

      const nome = p.getAttribute("data-nome") || 
                   p.querySelector(".competicao__participante-nome")?.textContent;
      if (nome) {
        participanteSelecionado = nome;
        if (confirmarNome) {
          confirmarNome.textContent = participanteSelecionado;
        }
      }
    });
  });

  if (btnVotarMain) {
    btnVotarMain.addEventListener("click", () => {
      if (!participanteSelecionado) {
        alert("Selecione um participante no grid antes de votar!");
        return;
      }
      const cardConfirmar = document.querySelector<HTMLElement>("#confirmar-voto");
      cardConfirmar?.scrollIntoView({ behavior: "smooth" });
    });
  }

  if (btnConfirmarVoto) {
    btnConfirmarVoto.addEventListener("click", () => {
      if (!participanteSelecionado) {
        alert("Selecione um participante no grid acima.");
        return;
      }
      alert(`Voto registrado com sucesso em "${participanteSelecionado}"!`);
    });
  }
}

async function init(): Promise<void> {
  await carregarDadosSanity();
  inicializarBotaoInscricao();
  inicializarInteratividadeVotacao();
}

if (document.readyState === "loading") {
  document.addEventListener("DOMContentLoaded", init);
} else {
  init();
}

import "./inscricao.css";
import { cmsFetch } from "../../services/cms";
import { usuarioAtual, fazerLogout } from "../../services/auth";
import { supabase } from "../../services/supabase";
import { ROTAS } from "../../config/routes";

const $ = <T extends HTMLElement>(id: string): T => {
  const el = document.getElementById(id);
  if (!el) {
    throw new Error(`Elemento #${id} não encontrado.`);
  }
  return el as T;
};

const slug = new URLSearchParams(window.location.search).get("slug");

const form = $("form-inscricao") as HTMLFormElement;
const inputVideo = $("video-url") as HTMLInputElement;
const btnInscrever = $("btn-inscrever") as HTMLButtonElement;
const msg = $("inscricao-msg") as HTMLParagraphElement;

function mostrarMsg(texto: string, ok = false): void {
  msg.textContent = texto;
  msg.classList.toggle("inscricao__msg--ok", ok);
  msg.hidden = false;
}

function irParaLogin(): void {
  const volta = encodeURIComponent(location.pathname + location.search);
  location.href = `${ROTAS.login}?aviso=login&redirect=${volta}`;
}

function linkValido(valor: string): boolean {
  try {
    const url = new URL(valor);
    return url.protocol === "http:" || url.protocol === "https:";
  } catch {
    return false;
  }
}

async function carregarCompeticao(): Promise<void> {
  const comp = await cmsFetch<{ titulo?: string }>(
    `*[_type == "competicao" && slug.current == $slug]{ titulo }`,
    { slug }
  );

  if (comp?.titulo) {
    $("competicao-nome").textContent = comp.titulo;
  }
}

async function carregarUsuario(): Promise<boolean> {
  const usuario = await usuarioAtual();
  if (!usuario) {
    irParaLogin();
    return false;
  }

  $("perfil-nome").textContent = usuario.email;
  return true;
}

async function enviar(e: Event): Promise<void> {
  e.preventDefault();
  msg.hidden = true;

  const video = inputVideo.value.trim();
  if (!video) {
    mostrarMsg("Cole o link do seu vídeo para se inscrever.");
    return;
  }

  if (!linkValido(video)) {
    mostrarMsg("O link do vídeo precisa começar com http:// ou https://.");
    return;
  }

  const usuario = await usuarioAtual();
  if (!usuario?.id) {
    irParaLogin();
    return;
  }

  if (!supabase) {
    mostrarMsg("Não foi possível conectar ao servidor. Tente novamente mais tarde.");
    return;
  }

  btnInscrever.disabled = true;
  const { error } = await supabase.from("inscricoes").insert({
    competicao_slug: slug,
    user_id: usuario.id,
    video_url: video,
  });

  if (error) {
    btnInscrever.disabled = false;
    mostrarMsg(
      error.code === "23505"
        ? "Você já está inscrito nesta competição."
        : "Não foi possível concluir a inscrição. Tente novamente em instantes."
    );
    return;
  }

  mostrarMsg("Inscrição enviada! Redirecionando…", true);
  setTimeout(() => {
    location.href = `${ROTAS.competicao}?slug=${encodeURIComponent(slug ?? "")}`;
  }, 1500);
}

async function init(): Promise<void> {
  if (!slug) {
    location.href = ROTAS.competicoes;
    return;
  }

  $("btn-voltar").setAttribute(
    "href",
    `${ROTAS.competicao}?slug=${encodeURIComponent(slug)}`
  );

  if (!(await carregarUsuario())) return;
  await carregarCompeticao();

  $("btn-relogin").addEventListener("click", async () => {
    await fazerLogout();
    irParaLogin();
  });

  form.addEventListener("submit", enviar);
}

if (document.readyState === "loading") {
  document.addEventListener("DOMContentLoaded", () => {
    void init();
  });
} else {
  void init();
}

import "./login.css";
import { checarAvisoLoginNaUrl } from "../../components/modal/modal";
import { fazerLogin, fazerCadastro } from "../../services/auth";
import { ROTAS } from "../../config/routes";

function modoCadastro(): boolean {
  return new URLSearchParams(window.location.search).get("modo") === "cadastro";
}

/**
 * Retorna o destino após o login.
 * Aceita apenas caminhos internos (começam com "/" e não com "//"),
 * evitando open redirect. Se não houver ?redirect= válido, vai para o perfil.
 */
function obterDestino(): string {
  const redirect = new URLSearchParams(window.location.search).get("redirect");

  if (redirect && redirect.startsWith("/") && !redirect.startsWith("//")) {
    return redirect;
  }
  return ROTAS.perfil;
}

function alternarInterfaceModo(): void {
  const eCadastro = modoCadastro();
  const tituloCard = document.getElementById("login-card-titulo");
  const textoCard = document.getElementById("login-card-texto");
  const linkCard = document.getElementById("login-card-link") as HTMLAnchorElement | null;
  const btnSubmit = document.querySelector<HTMLButtonElement>(".login__btn--submit");

  if (eCadastro) {
    document.title = "RYTH - Criar Conta";
    if (document.body) document.body.classList.add("login-page--cadastro");

    if (tituloCard) tituloCard.textContent = "Já tem uma conta?";
    if (textoCard) {
      textoCard.textContent =
        "Acesse sua conta para continuar acompanhando competições e interagindo com o ecossistema RYTH.";
    }
    if (linkCard) {
      linkCard.textContent = "Entrar na minha conta";
      linkCard.href = `${ROTAS.login}?modo=login`;
    }
    if (btnSubmit) btnSubmit.textContent = "cadastrar";
  } else {
    document.title = "RYTH - Login";
    if (document.body) document.body.classList.remove("login-page--cadastro");

    if (tituloCard) tituloCard.textContent = "Não tem uma conta?";
    if (textoCard) {
      textoCard.textContent =
        "Crie sua conta gratuitamente e comece a construir sua trajetória em um ecossistema que conecta dança, música e oportunidades.";
    }
    if (linkCard) {
      linkCard.textContent = "Criar minha conta";
      linkCard.href = `${ROTAS.login}?modo=cadastro`;
    }
    if (btnSubmit) btnSubmit.textContent = "entrar";
  }
}

function inicializarFormulario(): void {
  const form = document.querySelector<HTMLFormElement>(".login__form");
  if (!form) return;

  form.addEventListener("submit", async (e) => {
    e.preventDefault();

    const emailEl = document.getElementById("email") as HTMLInputElement | null;
    const senhaEl = document.getElementById("senha") as HTMLInputElement | null;

    const email = emailEl?.value.trim();
    const senha = senhaEl?.value.trim();

    if (!email || !senha) {
      alert("Por favor, preencha o e-mail e a senha.");
      return;
    }

    try {
      if (modoCadastro()) {
        const resultado = await fazerCadastro(email, senha);
        if (resultado.precisaConfirmar) {
          alert("Conta criada com sucesso! Verifique seu e-mail para confirmar a conta.");
          // Ainda não está logado: volta para o login em vez de ir ao perfil
          window.location.href = `${ROTAS.login}?modo=login`;
          return;
        }
        alert("Conta criada com sucesso!");
      } else {
        await fazerLogin(email, senha);
      }

      window.location.href = obterDestino();
    } catch (error) {
      const mensagem =
        error instanceof Error
          ? error.message
          : "Ocorreu um erro ao processar a solicitação. Tente novamente.";
      alert(mensagem);
    }
  });
}

async function init(): Promise<void> {
  alternarInterfaceModo();
  inicializarFormulario();

  checarAvisoLoginNaUrl();
}

if (document.readyState === "loading") {
  document.addEventListener("DOMContentLoaded", () => {
    void init();
  });
} else {
  void init();
}
