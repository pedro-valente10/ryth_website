import os

output_file = 'projeto_consolidado.txt'
ignore_dirs = {'node_modules', '.git', 'dist', 'build', '.vscode'}
ignore_exts = {'.png', '.jpg', '.jpeg', '.svg', '.ico', '.woff', '.woff2', '.ttf', '.lock'}

with open(output_file, 'w', encoding='utf-8') as outfile:
    for root, dirs, files in os.walk('.'):
        dirs[:] = [d for d in dirs if d not in ignore_dirs]
        for file in files:
            if file == output_file or file == 'package-lock.json' or any(file.endswith(ext) for ext in ignore_exts):
                continue
            path = os.path.join(root, file)
            outfile.write(f'========================================\n')
            outfile.write(f'ARQUIVO: {path}\n')
            outfile.write(f'========================================\n\n')
            try:
                with open(path, 'r', encoding='utf-8') as f:
                    outfile.write(f.read())
            except Exception:
                outfile.write('[Arquivo não lido ou binário]\n')
            outfile.write('\n\n')

print('Concluído! Arquivo gerado: projeto_consolidado.txt')

# RYTH Website

Site institucional da RYTH, construído com **Vite + TypeScript**, sem framework de UI (HTML/CSS/TS puro).

## Stack

- [Vite](https://vitejs.dev/) — bundler e dev server
- TypeScript
- HTML + CSS puro (sem framework de UI)

## Pré-requisitos

- Node.js (LTS mais recente recomendado)
- npm

## Configuração do ambiente

Clone o repositório e instale as dependências:

```bash
git clone <url-do-repositorio>
cd ryth_website
npm install
```

Rode o servidor de desenvolvimento:

```bash
npm run dev
```

A aplicação abre em `http://localhost:5173/`.

Build de produção:

```bash
npm run build
```

Os arquivos finais são gerados em `dist/`.

## Estrutura de pastas

```
ryth_website/
├── index.html                  # Página inicial (home) — entry point do Vite
├── public/                     # Arquivos estáticos servidos sem processamento
│   └── favicon.svg
├── src/
│   ├── main.ts                 # Entry point TypeScript, importa estilos e inicializa componentes globais
│   ├── vite-env.d.ts           # Referência de tipos do Vite (necessário p/ imports de .svg, .css, etc.)
│   ├── styles/
│   │   ├── reset.css
│   │   ├── global.css
│   │   └── variables.css       # Cores, fontes e outras variáveis CSS globais
│   ├── assets/
│   │   └── icons/
│   │       └── ryth-logo.svg   # Logo usada nos componentes (navbar, footer)
│   ├── components/
│   │   ├── navbar/
│   │   │   ├── navbar.ts       # Template + função renderNavbar()
│   │   │   └── navbar.css
│   │   └── footer/
│   │       ├── footer.ts       # Template + função renderFooter()
│   │       └── footer.css
│   └── pages/                  # Páginas internas do site (cada uma com seu próprio index.html)
├── tsconfig.json
└── vite.config.ts
```

## Convenções do projeto

### Componentes globais (Navbar / Footer)

Navbar e footer são **iguais em todas as páginas do site**. Em vez de duplicar HTML, cada um é uma função TypeScript que injeta seu template num container:

```html
<!-- em qualquer página -->
<div id="navbar-root"></div>
<!-- ...conteúdo da página... -->
<div id="footer-root"></div>
```

```ts
// no main.ts (ou entry point da página)
import { renderNavbar } from "./components/navbar/navbar";
import { renderFooter } from "./components/footer/footer";

renderNavbar();
renderFooter();
```

Essa abordagem foi escolhida por performance: o HTML já vem como string estática no `.ts`, sem `fetch` nem parsing assíncrono extra.

### CSS

Nomenclatura de classes em **BEM** (`bloco__elemento--modificador`), ex: `navbar__link--login`.

Variáveis globais (cores, fontes) ficam centralizadas em `src/styles/variables.css` e são consumidas via `var(--nome-da-variavel)` nos demais arquivos CSS.

### Assets: `src/assets/` vs `public/`

- **`src/assets/`** — para tudo que é referenciado dentro do código (importado em `.ts`/`.css`). O Vite processa, otimiza e faz cache-busting desses arquivos no build. Ex: a logo usada no navbar/footer.
- **`public/`** — para arquivos que precisam de uma URL fixa, sem processamento (ex: `favicon.svg`). Tudo aqui é copiado como está para a raiz do `dist` no build, e pode ser referenciado por caminho absoluto (`/favicon.svg`) de qualquer página.

### Múltiplas páginas

O projeto é multi-página: a home fica na raiz (`index.html`), e páginas internas devem ser criadas em `src/pages/<nome>/index.html`.

Ao criar uma **segunda** página, é necessário declarar todas as entradas no `vite.config.ts` (incluindo a da raiz) em `build.rollupOptions.input`, já que o Vite por padrão só builda o `index.html` da raiz automaticamente. Há um exemplo comentado no próprio `vite.config.ts`.

## Troubleshooting comum

- **`Cannot find module` em algum import**: confira se o caminho relativo bate com a localização real do arquivo (ex: `main.ts` está em `src/`, então imports dele para `src/components/...` devem ser `./components/...`, sem repetir `src/`).
- **Erro 404 de algum `.ts`/`.css` no navegador**: caminhos absolutos (`/algo`) no HTML são resolvidos a partir da raiz do projeto (onde fica o `vite.config.ts`), não da pasta onde está o HTML.
- **`Cannot find module '...svg'` no TypeScript**: confirme que existe `src/vite-env.d.ts` com `/// <reference types="vite/client" />`, e que o `tsconfig.json` tem `"include": ["src"]` (não só um arquivo específico).

function markActiveLink(): void {
  const normalize = (path: string): string =>
    path.replace(/\/index\.html$/, '/').replace(/\/$/, '') || '/';

  const current = normalize(location.pathname);

  document
    .querySelectorAll<HTMLAnchorElement>('.navbar__link a')
    .forEach((link) => {
      const href = normalize(new URL(link.href, location.origin).pathname);

      if (href === current) {
        link.setAttribute('aria-current', 'page');
      } else {
        link.removeAttribute('aria-current');
      }
    });
}

// Injeção dos componentes
renderNavbar();
markActiveLink();
renderFooter();
renderizarFAQ();

import { cmsFetch } from "../../services/cms";

type FaqItem = { pergunta: string; resposta: string };

export async function renderizarFAQ(): Promise<void> {
  const faqContainer = document.querySelector<HTMLElement>('.faq__list');
  if (!faqContainer) return;

  // Só busca os campos necessários e define a ordem
  const faqs = await cmsFetch<FaqItem[]>(
    `*[_type == "faq"] | order(_createdAt asc){ pergunta, resposta }`
  );

  // Se o Sanity falhar ou vier vazio, o HTML padrão da página permanece
  if (!faqs) return;

  faqContainer.replaceChildren(
    ...faqs.map((item) => {
      const li = document.createElement('li');
      li.className = 'faq__item';

      const h3 = document.createElement('h3');
      h3.className = 'faq__question';
      h3.textContent = item.pergunta;

      const p = document.createElement('p');
      p.className = 'faq__answer';
      p.textContent = item.resposta;

      li.append(h3, p);
      return li;
    })
  );
}

// src/services/cms.ts
import { sanityClient, urlFor } from './sanity'

// Retorna null em qualquer falha ou resultado vazio → o HTML padrão permanece
export async function cmsFetch<T = Record<string, any>>(
  query: string,
  params: Record<string, unknown> = {}
): Promise<T | null> {
  if (!sanityClient) return null
  try {
    const data = await sanityClient.fetch<T>(query, params)
    if (data == null || (Array.isArray(data) && data.length === 0)) return null
    return data
  } catch (err) {
    console.error('[CMS]', err)
    return null
  }
}

// Preenche o HTML com os dados:
//   data-cms="campo"                → textContent
//   data-cms="campo" data-cms-attr="href" → atributo
//   data-cms-img="campo"            → <img src> a partir de uma imagem do Sanity
//   data-cms-list="campo" + <template> → lista repetida
export function applyCms(root: ParentNode, data: Record<string, any>): void {
  root.querySelectorAll<HTMLElement>('[data-cms]').forEach((el) => {
    const value = data[el.dataset.cms!]
    if (typeof value !== 'string') return
    const attr = el.dataset.cmsAttr
    if (attr) el.setAttribute(attr, value)
    else el.textContent = value // textContent evita injeção de HTML
  })

  root.querySelectorAll<HTMLImageElement>('[data-cms-img]').forEach((img) => {
    const src = urlFor(data[img.dataset.cmsImg!])
    if (src) img.src = src
  })

  root.querySelectorAll<HTMLElement>('[data-cms-list]').forEach((container) => {
    const items = data[container.dataset.cmsList!]
    const tpl = container.querySelector('template')
    if (!Array.isArray(items) || items.length === 0 || !tpl) return

    Array.from(container.children).forEach((c) => c !== tpl && c.remove())
    items.forEach((item) => {
      const node = tpl.content.cloneNode(true) as DocumentFragment
      applyCms(node, item)
      container.appendChild(node)
    })
  })
}

// src/services/sanity.ts
import { createClient } from '@sanity/client'
import imageUrlBuilder from '@sanity/image-url'

const projectId = import.meta.env.VITE_SANITY_PROJECT_ID
const dataset = import.meta.env.VITE_SANITY_DATASET || 'production'

if (!projectId) {
  console.warn('[Sanity] VITE_SANITY_PROJECT_ID ausente: o site usará o conteúdo padrão do HTML.')
}

// null se não houver projectId, em vez de quebrar a importação
export const sanityClient = projectId
  ? createClient({
      projectId,
      dataset,
      useCdn: true, // requisições rápidas via CDN
      apiVersion: '2024-01-01',
    })
  : null

const builder = sanityClient ? imageUrlBuilder(sanityClient) : null

// Gera URLs das imagens cadastradas no Sanity (retorna '' se o Sanity não estiver configurado)
export function urlFor(source: any): string {
  return builder && source ? builder.image(source).auto('format').url() : ''
}
